   730→const PUBLIC_PATHS = ['/', '/login', '/lp', '/products', '/cart']
   731→
   732→function isPublicPath(path: string) {
   733→  if (path === '/' || path === '/login' || path === '/cart') return true
   734→  return PUBLIC_PATHS.some(p => p !== '/' && path.startsWith(p))
   735→}
   736→
   737→export function useApi() {
   738→  const req = $fetch.create({
   739→    onRequest({ options }) {
   740→      if (import.meta.client) {
   741→        const token = localStorage.getItem('ecos:token')
   742→        if (token) {
   743→          const headers = new Headers(options.headers as HeadersInit | undefined)
   744→          headers.set('Authorization', `Bearer ${token}`)
   745→          options.headers = headers
   746→        }
   747→      }
   748→    },
   749→    onResponseError({ response }) {
   750→      if (
   751→        response.status === 401
   752→        && import.meta.client
   753→        && !isPublicPath(window.location.pathname)
   754→      ) {
   755→        localStorage.removeItem('ecos:token')
   756→        navigateTo('/login')
   757→      }
   758→    },
   759→  })
   760→
   761→  return {
   762→    // ---- auth (§43) ----
