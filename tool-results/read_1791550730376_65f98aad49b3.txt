   968→export function useApi() {
   969→  const baseReq = $fetch.create({
   970→    onRequest({ options }) {
   971→      if (import.meta.client) {
   972→        const token = localStorage.getItem('ecos:token')
   973→        if (token) {
   974→          const headers = new Headers(options.headers as HeadersInit | undefined)
   975→          headers.set('Authorization', `Bearer ${token}`)
   976→          options.headers = headers
   977→        }
   978→      }
   979→    },
   980→  })
   981→
   982→  // ---- §43 token refresh (session continuity) ---------------------------
   983→  // Rotate a still-valid token for a fresh 12h lease. `current` lets the
   984→  // caller pass the exact credential that just failed (localStorage may
   985→  // already have been cleared by an earlier failure).
   986→  async function refreshToken(current?: string): Promise<string | null> {
   987→    if (!import.meta.client) return null
   988→    const token = current ?? localStorage.getItem('ecos:token')
   989→    if (!token) return null
   990→    try {
   991→      const res = await $fetch<LoginResponse>('/api/auth/refresh', {
   992→        method: 'POST',
   993→        headers: { Authorization: `Bearer ${token}` },
   994→      })
   995→      localStorage.setItem('ecos:token', res.token)
   996→      return res.token
   997→    }
   998→    catch {
   999→      return null
  1000→    }
  1001→  }
  1002→
  1003→  // Every admin API call funnels through here: on a 401 we try ONE silent
  1004→  // token rotation + retry (covers the "session older than 12h" case).
  1005→  // If the refresh also fails the session is genuinely dead -> clear + login.
  1006→  async function req<T>(url: string, opts: Record<string, unknown> = {}): Promise<T> {
  1007→    try {
  1008→      return await baseReq<T>(url as never, opts as never)
  1009→    }
  1010→    catch (err: unknown) {
  1011→      const status = (err as { status?: number; response?: { status?: number } })?.status
  1012→        ?? (err as { response?: { status?: number } })?.response?.status
  1013→      if (status === 401 && import.meta.client && !isPublicPath(window.location.pathname)) {
  1014→        const captured = localStorage.getItem('ecos:token') ?? undefined
  1015→        const fresh = await refreshToken(captured)
  1016→        if (fresh) return await baseReq<T>(url as never, opts as never)
  1017→        localStorage.removeItem('ecos:token')
  1018→        navigateTo('/login')
  1019→      }
  1020→      throw err
  1021→    }
  1022→  }
  1023→

  1032→      const token = localStorage.getItem('ecos:token')
  1033→      if (!token) return false
  1034→      try {
  1035→        const res = await $fetch<LoginResponse>('/api/auth/refresh', {
  1036→          method: 'POST', headers: { Authorization: `Bearer ${token}` },
  1037→        })
  1038→        localStorage.setItem('ecos:token', res.token)
  1039→        return true
  1040→      }
  1041→      catch { return false }
  1042→    },
