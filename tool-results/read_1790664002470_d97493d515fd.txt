    34→def _sales_snapshot(db: Session) -> dict[int, dict]:
    35→    """units + revenue per product across non-cancelled orders."""
    36→    rows = (
    37→        db.query(
    38→            om.OrderItem.product_id,
    39→            sa_func.sum(om.OrderItem.qty).label("units"),

    41→        )
    42→        .join(om.Order, om.Order.id == om.OrderItem.order_id)
    43→        .filter(om.Order.status.notin_(["draft", "cancelled", "failed"]))
    44→        .group_by(om.OrderItem.product_id)
    45→        .all()
    46→    )
    47→    return {r.product_id: {"units": int(r.units or 0), "revenue": float(r.revenue or 0.0)} for r in rows}

    63→def _gather_pricing(db: Session, params: dict) -> dict[str, Any]:
    64→    sales = _sales_snapshot(db)
    65→    products = db.query(cm.Product).filter(cm.Product.status == "active").all()
    66→    recs = []
    67→    for p in products:
    68→        s = sales.get(p.id, {"units": 0, "revenue": 0.0})
    69→        b = price_product(p.supplier_cost, "CNY", p.weight_kg, p.markup_pct)
    70→        current = b.ecos_price_ngn
    71→        if s["units"] == 0 and p.stock > 50:
    72→            factor, why = 0.92, "no sales yet with healthy stock — stimulate demand"
    73→        elif s["units"] >= 10:
    74→            factor, why = 1.06, "demand is proven (10+ units) — capture more margin"
    75→        else:
    76→            factor, why = 1.0, "early signal — hold price and observe"
    77→        suggested = round(current * factor / 5) * 5
    78→        cost = b.supplier_cost_ngn
    79→        base = cost + b.logistics_ngn + b.payment_cost_ngn + b.luxeen_margin_ngn
    80→        markup = (suggested - base) / cost if cost else p.markup_pct
    81→        recs.append({
    82→            "product_id": p.id, "product_title": p.title,
    83→            "current_price_ngn": current, "suggested_price_ngn": suggested,
    84→            "units_sold": s["units"], "stock": p.stock,
    85→            "margin_pct_current": round((current - base) / current * 100, 1) if current else 0,
    86→            "rationale": why,
    87→        })
    88→        # stash the inverse-computed markup for apply()
    89→        recs[-1]["_markup_pct"] = round(markup, 4)
    90→    return {"products": recs}

    93→def _heuristic_pricing(gathered: dict[str, Any]) -> dict[str, Any]:
    94→    recs = [{k: v for k, v in r.items() if not k.startswith("_")} for r in gathered["products"]]
    95→    changes = [r for r in recs if r["suggested_price_ngn"] != r["current_price_ngn"]]
    96→    return {
    97→        "analysis": "Margin + velocity scan across the active catalog.",
    98→        "recommendations": recs,
    99→        "summary": (
   100→            f"{len(changes)} of {len(recs)} products merit a price move; "
   101→            f"{len(recs) - len(changes)} hold steady."
   102→        ),
   103→    }

   106→def _apply_pricing(db: Session, operator, output: dict, actor: dict) -> dict[str, Any]:
   107→    applied = []
   108→    for r in output.get("recommendations", []):
   109→        p = db.get(cm.Product, r.get("product_id"))
   110→        suggested = r.get("suggested_price_ngn")
   111→        if p and suggested:
   112→            # recompute the markup that hits the approved price against
   113→            # CURRENT costs (they may have drifted since the run)
   114→            b = price_product(p.supplier_cost, "CNY", p.weight_kg, p.markup_pct)
   115→            if b.ecos_price_ngn == suggested:
   116→                continue  # no move proposed for this product
   117→            base = (b.supplier_cost_ngn + b.logistics_ngn
   118→                    + b.payment_cost_ngn + b.luxeen_margin_ngn)
   119→            if b.supplier_cost_ngn:
   120→                p.markup_pct = round((suggested - base) / b.supplier_cost_ngn, 4)
   121→                applied.append({"product_id": p.id, "old_price_ngn": b.ecos_price_ngn,
   122→                                "new_price_ngn": suggested})
   123→    return {"applied_changes": applied}

   130→def _gather_demand(db: Session, params: dict) -> dict[str, Any]:
   131→    sales = _sales_snapshot(db)
   132→    products = db.query(cm.Product).filter(cm.Product.status == "active").all()
   133→    out = []
   134→    for p in products:
   135→        supplier = db.get(sm.Supplier, p.supplier_id)
   136→        lead_weeks = max(1.0, (supplier.lead_time_days if supplier else 14) / 7.0)
   137→        units = sales.get(p.id, {"units": 0, "revenue": 0.0})["units"]
   138→        velocity = round(units / 4.0, 1)  # 4-week observed window
   139→        cover = round(p.stock / velocity, 1) if velocity else None
   140→        reorder = 0
   141→        if velocity:
   142→            need = lead_weeks * velocity * 1.5  # 1.5x safety stock
   143→            reorder = max(0, int(round(need - p.stock)))
   144→        out.append({
   145→            "product_id": p.id, "product_title": p.title,
   146→            "units_sold_4w": units, "weekly_velocity": velocity,
   147→            "weeks_of_cover": cover, "supplier_lead_time_weeks": round(lead_weeks, 1),
   148→            "stock": p.stock, "suggested_reorder_qty": reorder,
   149→            "risk": ("stockout" if cover is not None and cover < lead_weeks
   150→                     else "healthy" if cover is None or cover > 6 else "watch"),
   151→        })
   152→    return {"products": out}

   353→        return float(re.sub(r"[^0-9.]", "", str(v)) or default)

   358→def _parse_listing(text: str) -> dict[str, Any]:
   359→    """Parse a raw supplier listing: 'Title | 95 CNY | 0.25kg | electronics | k=v; k=v'."""
   360→    lines = [ln.strip() for ln in text.strip().splitlines() if ln.strip()]
   361→    parts = re.split(r"\s*\|\s*", lines[0]) if lines else []
   362→    title = (parts[0] if parts else "Untitled import").strip()[:255]
   363→    cost = _to_num_or(parts[1] if len(parts) > 1 else 0, 0.0)
   364→    currency = "CNY"
   365→    for token in parts[1:]:
   366→        up = token.upper()
   367→        if "USD" in up or "$" in up:
   368→            currency = "USD"
   369→        elif "CNY" in up or "¥" in up or "RMB" in up:
   370→            currency = "CNY"
   371→    weight = next((_to_num_or(t, 0.5) for t in parts if "kg" in t.lower()), 0.5)
   372→    category = "general"
   373→    if len(parts) > 3:
   374→        category = parts[3].strip().lower().replace(" ", "-") or "general"
   375→    specs: dict[str, str] = {}
   376→    specs_raw = next((t for t in parts if "=" in t and ";" in t), "")
   377→    if specs_raw:
   378→        for pair in specs_raw.split(";"):
   379→            if "=" in pair:
   380→                k, _, v = pair.partition("=")
   381→                specs[k.strip()] = v.strip()
   382→    return {"title": title, "cost": cost, "currency": currency,
   383→            "weight_kg": weight, "category": category, "specs": specs}

   508→def _now_utc():
   509→    """Naive UTC (SQLite stores naive datetimes — keep comparisons consistent)."""
   510→    from datetime import datetime as _dt, timezone as _tz
   511→    return _dt.now(_tz.utc).replace(tzinfo=None)

   513→
   514→def _gather_logistics_ops(db: Session, params: dict) -> dict[str, Any]:
   515→    from app.logistics import models as lm_
   516→
   517→    stall_hours = max(6, int(params.get("stall_hours") or 48))
   518→    now = _now_utc()
   519→    shipments = db.query(lm_.Shipment).order_by(lm_.Shipment.id.desc()).limit(200).all()
   520→    active_states = ("processing", "in_transit", "out_for_delivery")
   521→    rows, stalled = [], []
   522→    for s in shipments:
   523→        last = (
   524→            db.query(lm_.TrackingEvent)
   525→            .filter(lm_.TrackingEvent.shipment_id == s.id)
   526→            .order_by(lm_.TrackingEvent.occurred_at.desc(), lm_.TrackingEvent.id.desc())
   527→            .first()
   528→        )
   529→        age_h = None
   530→        if last and last.occurred_at:
   531→            age_h = round((now - last.occurred_at).total_seconds() / 3600, 1)
   532→        row = {"shipment_id": s.id, "order_id": s.order_id, "tracking_code": s.tracking_code,
   533→               "carrier": s.carrier, "status": s.status,
   534→               "last_checkpoint": last.code if last else None,
   535→               "hours_since_checkpoint": age_h}
   536→        rows.append(row)
   537→        if s.status in active_states and (age_h is None or age_h >= stall_hours):
   538→            stalled.append(row)
   539→    delivered_rows = [r for r in rows if r["status"] == "delivered"]
   540→    return {
   541→        "threshold_hours": stall_hours,
   542→        "totals": {"tracked": len(rows), "delivered": len(delivered_rows),
   543→                   "active": len(rows) - len(delivered_rows), "stalled": len(stalled)},
   544→        "stalled_shipments": stalled,
   545→    }

   563→        "summary": (

   624→def _heuristic_analyst(gathered: dict[str, Any]) -> dict[str, Any]:
   625→    led = gathered["ledger_totals_by_type"]
   626→    revenue = led.get("customer_payment", 0.0)
   627→    refunds = led.get("refund", 0.0)
   628→    supplier = abs(led.get("supplier_payable", 0.0))
   629→    logistics = abs(led.get("logistics_cost", 0.0))
   630→    paycost = abs(led.get("payment_cost", 0.0))
   631→    luxeen = abs(led.get("luxeen_economics", 0.0))
   632→    operator = led.get("operator_economics", 0.0)
   633→    net = revenue + refunds  # refunds arrive pre-signed (negative)
   634→    contribution = net - supplier - logistics - paycost - luxeen
   635→
   636→    orders = gathered["orders"]
   637→    delivery_rate = round(orders["delivered"] / orders["total"], 3) if orders["total"] else 0.0
   638→    rma = gathered["returns"]
   639→    return_rate = round(rma["rma_count"] / orders["total"], 3) if orders["total"] else 0.0
   640→    unsettled = gathered["unsettled_obligations"]
   641→
   642→    risks: list[str] = []
   643→    if delivery_rate < 0.5 and orders["total"] >= 5:
   644→        risks.append(f"Delivery rate is {delivery_rate:.0%} — cash is stuck in the pipeline; chase the in-flight orders.")
   645→    if return_rate > 0.15:
   646→        risks.append(f"Return rate {return_rate:.0%} is above the 15% comfort line — inspect the top-returned SKUs.")
   647→    if unsettled["amount"] > 0:
   648→        risks.append(f"₦{unsettled['amount']:,.0f} across {unsettled['entries']} ledger entries is still unsettled — counterparties are waiting.")
   649→    if not risks:
   650→        risks.append("No red flags: pipeline, returns and settlements all look healthy this period.")
   651→
   652→    highlights = [
   653→        f"Gross collections ₦{revenue:,.0f}; net after refunds ₦{net:,.0f}.",
   654→        f"Contribution after supplier/logistics/fees/Luxeen: ₦{contribution:,.0f} "
   655→        f"({(contribution / net * 100 if net else 0):.1f}% of net).",
   656→        f"Operator economics so far: ₦{operator:,.0f}.",
   657→    ]
   658→    return {
   659→        "highlights": highlights,
   660→        "risks": risks,
   661→        "recommendations": [
   662→            "Reconcile the settlement queue weekly — stale payables erode supplier trust and lane priority.",
   663→            "Watch the stockout-risk SKUs; a stockout on a proven mover costs more than a PO.",
   664→            "Keep COD confirmations fast: every hour between order and confirmation raises the cancel rate.",
   665→        ],
   666→        "metrics": {
   667→            "net_revenue_ngn": round(net, 2), "contribution_ngn": round(contribution, 2),
   668→            "delivery_rate": delivery_rate, "return_rate": return_rate,
   669→            "unsettled_ngn": unsettled["amount"],
   670→        },
   671→        "summary": f"Net ₦{net:,.0f}, contribution ₦{contribution:,.0f}, {len(risks)} risk note(s).",
   672→    }
